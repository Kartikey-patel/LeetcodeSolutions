class Solution {
private:
    bool dfs(vector<int>&vis, vector<int>&pathVis, vector<int>& safeNode, vector<vector<int>>& graph, int node){
        vis[node] = 1;
        pathVis[node] = 1;

        for(auto it : graph[node]){
            if(!vis[it]){
                if(dfs(vis,pathVis,safeNode,graph,it) == false) return false;
            }else if(pathVis[it]){
                return false;
            }
        }
        pathVis[node] = 0;
        safeNode[node] = 1;
        return true;
    }
